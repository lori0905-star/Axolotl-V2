function spawnBubble(){

  if (!bubbleActive || !tankBuilt) return;

  const app = $('#app');

  const rect = app.getBoundingClientRect();

  if (!rect.width || !rect.height) return;


  // 随机位置：水面附近 + 随机高度，偏向右侧（水面区域）

  const leftPct = rand(8, 88);

  const size = rand(32, 56);

  const bubble = document.createElement('div');

  bubble.className = 'bubble';

  bubble.style.width = size + 'px';

  bubble.style.height = size + 'px';

  bubble.style.left = leftPct + '%';

  bubble.style.bottom = rand(10, 30) + '%';


  // 随机漂浮方向

  const driftX = rand(-25, 25);

  const riseDuration = rand(5, 9);

  const driftDuration = rand(3, 5);


  bubble.style.setProperty('--drift-x', driftX + 'px');

  bubble.style.animation = `

    bubbleFloatUp ${riseDuration}s linear forwards,

    bubbleDrift ${driftDuration}s ease-in-out infinite alternate

  `;


  // 点击戳破

  bubble.addEventListener('click', (e) => {

    e.stopPropagation();

    popBubble(bubble, e);

  });


  // 触摸优化

  bubble.addEventListener('touchstart', (e) => {

    e.preventDefault();

    e.stopPropagation();

    popBubble(bubble, e.touches[0]);

  }, { passive: false });


  app.appendChild(bubble);


  // 自动消失

  setTimeout(() => {

    if (bubble.parentNode) {

      bubble.style.transition = 'opacity .8s';

      bubble.style.opacity = '0';

      setTimeout(() => bubble.remove(), 800);

    }

  }, riseDuration * 1000);

}


function popBubble(bubbleEl, event){

  if (bubbleEl.classList.contains('pop')) return;

  bubbleEl.classList.add('pop');


  // 随机奖励

  const reward = BUBBLE_REWARDS[Math.floor(Math.random() * BUBBLE_REWARDS.length)];

  const rect = bubbleEl.getBoundingClientRect();

  const appRect = $('#app').getBoundingClientRect();

  const x = rect.left - appRect.left + rect.width/2;

  const y = rect.top - appRect.top + rect.height/2;


  // 奖励飘字

  const span = document.createElement('span');

  span.className = 'bubble-reward';

  span.textContent = reward;

  span.style.left = x + 'px';

  span.style.top = y + 'px';

  span.style.fontSize = rand(22, 36) + 'px';

  $('#app').appendChild(span);

  setTimeout(() => span.remove(), 1000);


  // 小振动 + 音效感（无声音，用视觉）

  vibrate(15);


  // 把零食“收入囊中”—— 显示提示

  if (Math.random() < 0.4) {

    showBubble('谢谢你的小零食！');

  }


  // 延迟移除泡泡

  setTimeout(() => {

    if (bubbleEl.parentNode) bubbleEl.remove();

  }, 400);

}


function startBubbleSystem(){

  if (bubbleInterval) clearInterval(bubbleInterval);

  // 每 2.5~5 秒随机生成一个泡泡

  function scheduleNext(){

    const delay = rand(2500, 5000);

    bubbleInterval = setTimeout(() => {

      if (bubbleActive && tankBuilt) {

        spawnBubble();

        // 偶尔多生成一个

        if (Math.random() < 0.3) setTimeout(spawnBubble, 400);

      }

      scheduleNext();

    }, delay);

  }

  scheduleNext();

}

/* 把屏幕坐标转换成 SVG 内部坐标（考虑 slice 裁切） */

function svgPoint(e, svgEl){

  const rect = svgEl.getBoundingClientRect();

  if (!rect.width || !rect.height) return null;

  const vbW = 400, vbH = 700;

  const rx = rect.width / vbW, ry = rect.height / vbH;

  const r = Math.max(rx, ry);             // slice => 取大的比例

  const drawnW = vbW * r, drawnH = vbH * r;

  const offX = (rect.width  - drawnW) / 2;

  const offY = (rect.height - drawnH) / 2;

  const x = (e.clientX - rect.left - offX) / r;

  const y = (e.clientY - rect.top  - offY) / r;

  return { x, y };

}

let lastPetPick = 0;

function tick(now){

  axoState.bob = Math.sin(now * 0.0018) * 7;

  axoState.wiggle *= 0.90;

  axoState.squash *= 0.88;

  const sc = axoScale() * (1 + axoState.squash * 0.18);


  /* 食物优先 */

  const uneaten = activeFoods.filter(f => !f.eaten);

  if (uneaten.length){

    let best = null, bd = Infinity;

    uneaten.forEach(f => {

      const d = Math.hypot(f.x - axoState.x, f.y - axoState.y);

      if (d < bd){ bd = d; best = f; }

    });

    if (best){

      axoState.tx = clamp(best.x - 18, 110, 300);

      axoState.ty = clamp(best.y - 4, 200, 545);

      if (Math.abs(best.x - axoState.x) > 5) axoState.facing = best.x < axoState.x ? -1 : 1;

    }

  }

  /* 手指追踪（其次） */

  else if (pointerFollow.active && now < pointerFollow.until){

    const tx = clamp(pointerFollow.x - 32, 90, 310);

    const ty = clamp(pointerFollow.y, 190, 555);

    axoState.tx = tx; axoState.ty = ty;

    if (Math.abs(tx - axoState.x) > 8) axoState.facing = tx < axoState.x ? -1 : 1;

  } else if (pointerFollow.active){

    pointerFollow.active = false;

  }


  axoState.x += (axoState.tx - axoState.x) * 0.016;

  axoState.y += (axoState.ty - axoState.y) * 0.016;


  const wig = axoState.wiggle * Math.sin(now * 0.05) * 6;


  if (mainAxoEl){

    mainAxoEl.setAttribute('transform',

      `translate(${(axoState.x + wig).toFixed(2)},${(axoState.y + axoState.bob).toFixed(2)}) scale(${(sc*axoState.facing).toFixed(3)},${sc.toFixed(3)})`);

  }


  /* 伴侣 / 宝宝 */

  const allPets = [...companionEls, ...babyEls];

  allPets.forEach((p, i) => {

    if (now - lastPetPick > 3000 + i*900){

      p.tx = rand(100, 300); p.ty = rand(240, 500);

      if (p.tx < p.x - 6) p.facing = -1;

      else if (p.tx > p.x + 6) p.facing = 1;

    }

    p.x += (p.tx - p.x) * 0.009;

    p.y += (p.ty - p.y) * 0.009;

    const b = Math.sin(now * 0.002 + i * 1.7) * 6;

let sudokuState = null;

function shuffle(a){ for (let i=a.length-1;i>0;i--){ const j=Math.floor(Math.random()*(i+1)); [a[i],a[j]]=[a[j],a[i]]; } return a; }

function isValid(b,r,c,n){

  for (let i=0;i<9;i++) if (b[r][i]===n || b[i][c]===n) return false;

  const br=Math.floor(r/3)*3, bc=Math.floor(c/3)*3;

  for (let i=0;i<3;i++) for (let j=0;j<3;j++) if (b[br+i][bc+j]===n) return false;

  return true;

}

function fillBoard(b,pos=0){

  if (pos===81) return true;

  const r=Math.floor(pos/9), c=pos%9;

  for (const n of shuffle([1,2,3,4,5,6,7,8,9])){

    if (isValid(b,r,c,n)){ b[r][c]=n; if (fillBoard(b,pos+1)) return true; b[r][c]=0; }

  }

  return false;

}

function generateSolution(){ const b=Array.from({length:9},()=>Array(9).fill(0)); fillBoard(b); return b; }

function makePuzzle(sol,holes){

  const p=sol.map(r=>[...r]);

  const idx=shuffle([...Array(81).keys()]);

  for (let i=0;i<holes;i++){ const r=Math.floor(idx[i]/9), c=idx[i]%9; p[r][c]=0; }

  return p;

}

function startSudoku(){

  const solution=generateSolution();

  const puzzle=makePuzzle(solution,36);

  sudokuState={ solution, puzzle, user:puzzle.map(r=>[...r]), selected:null, done:false };

  renderSudoku(); renderPad();

  showScene('sudoku');

}

function renderSudoku(){

  const board=$('#sudokuBoard'); board.innerHTML='';

  for (let r=0;r<9;r++) for (let c=0;c<9;c++){

    const cell=document.createElement('div');

    cell.className='sdk-cell';

    const fixed=sudokuState.puzzle[r][c]!==0;

    const v=sudokuState.user[r][c];

    if (fixed) cell.classList.add('fixed');

    if (v) cell.textContent=v;

    if (c===2||c===5) cell.classList.add('br');

    if (r===2||r===5) cell.classList.add('bb');

    if (!fixed && v && v!==sudokuState.solution[r][c]) cell.classList.add('wrong');

    if (sudokuState.selected && sudokuState.selected.r===r && sudokuState.selected.c===c) cell.classList.add('sel');

    cell.addEventListener('click',()=>selectCell(r,c));

    board.appendChild(cell);

  }

}

function selectCell(r,c){

  if (!sudokuState||sudokuState.done) return;

  if (sudokuState.puzzle[r][c]!==0) return;

  sudokuState.selected={r,c}; renderSudoku();

}

function inputNumber(n){

  if (!sudokuState||sudokuState.done) return;

  const sel=sudokuState.selected; if (!sel) return;

  const {r,c}=sel; if (sudokuState.puzzle[r][c]!==0) return;

  sudokuState.user[r][c]=n; renderSudoku(); vibrate(8); checkComplete();

}

function checkComplete(){

  for (let r=0;r<9;r++) for (let c=0;c<9;c++)

    if (sudokuState.user[r][c]!==sudokuState.solution[r][c]) return;

  sudokuState.done=true;

  vibrate([40,60,40]);

  toast('数独完成！<br>今天有满满一份甜品 🍰');

}
